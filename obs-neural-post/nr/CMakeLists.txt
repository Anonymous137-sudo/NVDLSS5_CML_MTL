cmake_minimum_required(VERSION 3.28)
project(obs-dlss-nr LANGUAGES C CXX OBJC)
set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0)
set(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(DLSS_OBS_NR_DIR "${PROJECT_ROOT}/../obs-dlss5-nr" CACHE PATH "obs-dlss5-nr macOS port checkout")
add_library(obs-dlss-nr MODULE obs-dlss-nr.m MacDisplayCapture.m DLSSControls.m "${DLSS_OBS_NR_DIR}/src/dlss5-nr-filter.cpp")
target_compile_features(obs-dlss-nr PRIVATE cxx_std_20)
target_include_directories(obs-dlss-nr PRIVATE "${DLSS_OBS_NR_DIR}/src")
if(NOT TARGET OBS::libobs)
  set(OBS_HEADERS "${PROJECT_ROOT}/.deps/obs-studio-32.1.0" CACHE PATH "OBS release source headers")
  set(OBS_APP "/Applications/OBS.app" CACHE PATH "Installed stock OBS application")
  if(NOT EXISTS "${OBS_HEADERS}/libobs/obs.h")
    message(FATAL_ERROR "Set OBS_HEADERS to the extracted OBS release source")
  endif()
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/obsconfig.h"
    "#pragma once\n#define OBS_RELEASE_CANDIDATE 0\n#define OBS_BETA 0\n")
  target_include_directories(obs-dlss-nr PRIVATE "${OBS_HEADERS}/libobs"
    "${OBS_HEADERS}/frontend/api" "${CMAKE_CURRENT_BINARY_DIR}")
endif()
target_include_directories(obs-dlss-nr PRIVATE "${PROJECT_ROOT}/.deps/simde-0.8.2")
target_compile_options(obs-dlss-nr PRIVATE $<$<COMPILE_LANGUAGE:OBJC>:-fobjc-arc> -fmodules -Wall -Wextra)
if(TARGET OBS::libobs)
  target_link_libraries(obs-dlss-nr PRIVATE OBS::libobs obs-frontend-api)
else()
  target_link_libraries(obs-dlss-nr PRIVATE "${OBS_APP}/Contents/Frameworks/libobs.framework/libobs" "${OBS_APP}/Contents/Frameworks/obs-frontend-api.dylib")
endif()
target_link_libraries(obs-dlss-nr PRIVATE "${PROJECT_ROOT}/build/nr/libnvidia_nr.dylib" "${PROJECT_ROOT}/build/nr/libobs_dlss5_nr.dylib" "-framework ScreenCaptureKit" "-framework AVFoundation" "-framework VideoToolbox" "-framework CoreVideo" "-framework IOSurface" "-framework Foundation" "-framework AppKit" "-framework QuartzCore" "-framework CoreText")
set_target_properties(obs-dlss-nr PROPERTIES PREFIX "" SUFFIX "" BUNDLE TRUE BUNDLE_EXTENSION plugin MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in" BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@loader_path/../Frameworks;@executable_path/../Frameworks")

if(TARGET OBS::libobs)
  set_target_properties_obs(obs-dlss-nr PROPERTIES FOLDER plugins)
endif()

set_target_properties(obs-dlss-nr PROPERTIES XCODE_ATTRIBUTE_MACOSX_DEPLOYMENT_TARGET 14.0)
